# Runtime dependencies pinned for reproducible production builds.
# Keep this list as small as possible — the app uses stdlib for most things.
# pypdf — upgrade range covers CVE-2025-55197 / CVE-2025-62707 /
# CVE-2025-62708 / CVE-2025-66019 and the 2026 CVE series through
# CVE-2026-41314. Verified by pip-audit (--strict) in CI.
pypdf>=6.10.2,<7.0.0
pywebpush==1.14.0
# cryptography backs the ChaCha20-Poly1305 AEAD encryption-at-rest layer
# in company_discovery/crypto_kit.py and the AI-Act audit-log salt
# hashing in company_discovery/audit_log.py. It is also a transitive
# dependency of pywebpush; we pin it explicitly so the wheel selected
# is under our control rather than the resolver's, and so future code
# does not silently depend on a transitive dep that pywebpush could
# in principle drop.
#
# Modern cryptography (42+) ships pre-built wheels for macOS x86_64 +
# arm64, Linux x86_64 + aarch64 (manylinux2014 + musllinux), and
# Windows. This range eliminates the historical cffi source-build
# failure reported on minimal Linux containers without rust +
# build-essential — those environments now resolve to a pre-built
# wheel and the install succeeds without compilation. Verified via
# the fresh-clone-install workflow (.github/workflows/fresh-clone-
# install.yml).
cryptography>=42.0.0,<50.0.0
# jsonschema enforces MCP tools/call inputSchema validation before dispatch
# (mcp_server.py). Light dependency, well-maintained, no native code.
jsonschema==4.25.1
# PyJWT backs the OIDC ID-token signature verification in
# company_discovery/sso_oidc.py (13-plan item 12/13). RS256 + ES256 are
# the algorithms every major IdP (Okta, Azure AD, Google Workspace,
# Keycloak, Auth0) ships by default. Implementing JWT verification by
# hand against arbitrary IdP key sets is security-critical — PyJWT
# is the industry standard. Builds on the existing cryptography
# dependency (no new transitive surface). Active maintenance + audited
# security history; the Python community's de-facto JWT layer.
PyJWT[crypto]>=2.8.0,<3.0.0
# signxml backs the SAML 2.0 SP layer in company_discovery/sso_saml.py
# (13-plan item 12/13). SAML's XML Signature spec requires
# canonicalisation (XML-C14N) which is notoriously fragile —
# implementing it from scratch is the most common security failure
# point in DIY SAML implementations. signxml is pure-Python (depends
# on lxml + the cryptography we already have), actively maintained,
# and built specifically for SAML / XML DSig use cases. Choosing it
# over python3-saml because python3-saml depends on xmlsec1 + libxml2
# C libraries, complicating reproducible builds + Nix packaging.
signxml>=4.0.0,<5.0.0

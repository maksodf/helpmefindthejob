# Development-only dependencies for DirectJob Scout.
# These are NOT required to run the application — only to contribute
# code. The runtime list is in requirements.txt; install both during
# contributor setup:
#
#   pip install -r requirements.txt
#   pip install -r requirements-dev.txt
#   pre-commit install
#
# See CONTRIBUTING.md for the full setup sequence.

# Pre-commit framework. Runs SPDX-header check, ruff (lint + format-
# check), and codespell on staged Python files. Full hook list in
# .pre-commit-config.yaml.
pre-commit>=3.5.0,<5.0.0

# Ruff — Python linter + formatter. Single tool replaces flake8 +
# isort + pyupgrade + black for the lint and format passes. Config
# in pyproject.toml [tool.ruff].
ruff>=0.6.0,<1.0.0

# Mypy — static type checker. Permissive starter config with
# strict-only overrides for the modules with clear contracts
# (audit_log, crypto_kit, mcp_server). Config in pyproject.toml
# [tool.mypy].
mypy>=1.10.0,<2.0.0
types-jsonschema>=4.17.0
types-requests>=2.31.0

# Coverage — line + branch coverage. Config in pyproject.toml
# [tool.coverage].
coverage[toml]>=7.4.0,<8.0.0

# Codespell — typo finder. Config in pyproject.toml [tool.codespell].
codespell>=2.2.0,<3.0.0

# Pip-audit — dependency vulnerability scanner. Run in CI on every
# push.
pip-audit>=2.7.0,<3.0.0

# mkdocs-material — static documentation site generator that produces
# the GitHub Pages site at https://maksodf.github.io/directjob-scout/.
# Built locally with `mkdocs build --strict` and deployed by
# `.github/workflows/docs-publish.yml` on every push. Config lives in
# `mkdocs.yml` at the repo root. Major pin keeps the theme contract
# stable across the 4-week grant sprint; the upgrade conversation
# happens deliberately, not by surprise.
mkdocs-material>=9.5.0,<10.0.0
